\section{Mathematical Models}
\begin{frame}{Mathematical Models: Physics}
    \textbf{Vehicle Kinematics:}
    \begin{itemize}
        \item Velocity Update: $v = v_0 + a \cdot \Delta t$
        \item Position Update: $\overrightarrow{p} = \overrightarrow{p}_0 + \overrightarrow{d} \cdot (v \cdot \Delta t)$
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Braking Model (IDM Variant):}
    \begin{itemize}
        \item Required deceleration to stop at distance $d$:
        \item $a_{brake} = -\frac{v^2}{2 \cdot d}$
    \end{itemize}
\end{frame}

\begin{frame}{Mathematical Models: Collision Detection}
    \textbf{Line Intersection using Cramer's Rule:}
    \begin{itemize}
        \item Determinant: $D = (x_1 - x_2)(y_3 - y_4) - (y_1 - y_2)(x_3 - x_4)$
        \item Coordinates: 
        \begin{align*}
            x &= \frac{D_1(x_3 - x_4) - (x_1 - x_2)D_2}{D} \\
            y &= \frac{D_1(y_3 - y_4) - (y_1 - y_2)D_2}{D}
        \end{align*}
    \end{itemize}
