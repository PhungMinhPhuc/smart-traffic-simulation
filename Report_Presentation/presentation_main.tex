\section{Introduction}
\begin{frame}{Introduction}
    \begin{itemize}
        \item \textbf{Context:} Rapid urbanization leads to severe traffic congestion.
        \item \textbf{Objective:} Develop a Java-based microscopic smart traffic simulation.
        \item \textbf{Features:}
        \begin{itemize}
            \item Realistic vehicle kinematics and driver behaviors.
            \item Dynamic traffic networks (lights, intersections, multi-lane roads).
            \item OOP principles for modularity and scalability.

\section{System Architecture}
\begin{frame}{System Architecture: Map Components}
    \begin{table}
        \centering
        \begin{tabular}{|p{0.32\linewidth} |p{0.60\linewidth}|}
            \hline
            \textbf{Element} & \textbf{Functionality} \\
            \hline
            \texttt{TrafficMap} & Stores the network graph and manages Roads, Lanes, and TrafficNodes. \\\hline
            \texttt{TrafficNode} & Controls crossing logic and traffic light coordination. \\\hline
            \texttt{Path} & Paths for vehicle to travel inside TrafficNodes\\\hline
            \texttt{Road} & Defines road geometry and connectivity between TrafficNodes, contains left and right way\\\hline
            \texttt{Way} &  Left or right way of Road, contains Lanes and a TrafficLight\\\hline
            \texttt{TrafficLight} & Controls signal state, countdown timer, and lane stopping behavior. \\\hline
            \texttt{Lane} & Represents lane-level Paths, direction, and capacity. \\
            \hline
        \end{tabular}
    \end{table}
\end{frame}

\begin{frame}{System Architecture: Vehicle Logic}
    \begin{table}

            \texttt{Vehicle} & Simulated Vehicle with position, velocity, and kinematic updates.\\\hline
 Car, Bike, Ambulance, ....&Specific vehicles inherit from Vehicle\\\hline
            \texttt{DriverBehavior} & Encapsulates driving policies (aggressive, normal, cautious, ...).\\
